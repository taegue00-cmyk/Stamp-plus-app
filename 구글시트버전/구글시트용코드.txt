//  이 글 전체를 구글 시트의 Apps Script 편집기에 붙여 넣어 주세요.
//  자세한 순서는 '설정방법.md'를 보세요.

var CFG = {
  PIN: '0000',               // ← 사장님 비밀번호(숫자 4자리)로 바꿔 주세요. (0000, 1234 같은 쉬운 숫자는 피하세요)
  GOAL: 10,                  // 스탬프를 몇 개 모으면 보상?
  REWARD: '무료 음료 1잔',    // 보상 이름
  TZ: 'Asia/Seoul'           // 시간대 (바꾸지 마세요)
};

var SHEET_CUSTOMER = '손님';
var SHEET_LOG = '기록';

// ------------------------------------------------------------
// 처음 한 번만 실행하세요. (시트 만들기 + 비밀번호 저장)
// 비밀번호를 잊었을 때도, 위 PIN을 바꾸고 이 함수를 다시 실행하면 새 비밀번호로 바뀌어요.
// ------------------------------------------------------------
function 처음설정() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();

  var c = ss.getSheetByName(SHEET_CUSTOMER) || ss.insertSheet(SHEET_CUSTOMER);
  if (c.getLastRow() === 0) c.appendRow(['뒷자리', '별명', '스탬프', '받은 선물', '마지막 적립']);
  c.getRange('A:A').setNumberFormat('@');            // 0으로 시작하는 번호가 안 깨지게 글자로 저장
  c.getRange('E:E').setNumberFormat('yyyy-mm-dd hh:mm:ss');

  var l = ss.getSheetByName(SHEET_LOG) || ss.insertSheet(SHEET_LOG);
  if (l.getLastRow() === 0) l.appendRow(['시각', '뒷자리', '행동', '그 뒤 스탬프 수']);
  l.getRange('A:A').setNumberFormat('yyyy-mm-dd hh:mm:ss');
  l.getRange('B:B').setNumberFormat('@');

  var props = PropertiesService.getScriptProperties();
  if (/^\d{4}$/.test(CFG.PIN)) {
    props.setProperty('PIN', CFG.PIN);
    props.setProperty('FAILS', '0');
    props.deleteProperty('LOCK_UNTIL');
  }
  if (!props.getProperty('PIN')) {
    throw new Error('CFG의 PIN을 숫자 4자리로 바꾼 뒤 다시 실행해 주세요.');
  }
  Logger.log('설정 완료!');
}

// ------------------------------------------------------------
// 인터넷에서 들어오는 요청을 받는 곳
// ------------------------------------------------------------
function doGet() {
  return ContentService.createTextOutput('브릿지 스탬프 서버가 잘 켜져 있어요.');
}

function doPost(e) {
  var out;
  try {
    var req = JSON.parse(e.postData.contents);
    out = handle(req);
  } catch (err) {
    out = { ok: false, code: 'ERROR' };
  }
  return ContentService.createTextOutput(JSON.stringify(out)).setMimeType(ContentService.MimeType.JSON);
}

function handle(req) {
  var action = String(req.action || '');

  // 손님 조회: 누구나 가능, 스탬프 개수만 알려 줘요 (별명은 감춰요)
  if (action === 'lookup') {
    var l4 = String(req.last4 || '');
    if (!/^\d{4}$/.test(l4)) return { ok: false, code: 'BAD_INPUT' };
    var sh = getSheet(SHEET_CUSTOMER);
    var row = findRow(sh, l4);
    return { ok: true, found: row > 0, stamps: row ? Number(sh.getRange(row, 3).getValue()) || 0 : 0, goal: CFG.GOAL, reward: CFG.REWARD };
  }

  if (['login', 'add', 'undo', 'redeem', 'setpin'].indexOf(action) < 0) return { ok: false, code: 'BAD_INPUT' };

  var lock = LockService.getScriptLock();     // 동시에 눌러도 한 명씩 차례로 처리
  lock.waitLock(10000);
  try {
    var bad = checkPin(req.pin);
    if (bad) return bad;
    var props = PropertiesService.getScriptProperties();

    if (action === 'login') return { ok: true, goal: CFG.GOAL, reward: CFG.REWARD, today: todayCount(0) };

    if (action === 'setpin') {
      var np = String(req.newPin || '');
      if (!/^\d{4}$/.test(np)) return { ok: false, code: 'BAD_INPUT' };
      props.setProperty('PIN', np);
      return { ok: true };
    }

    var last4 = String(req.last4 || '');
    if (!/^\d{4}$/.test(last4)) return { ok: false, code: 'BAD_INPUT' };
    var sheet = getSheet(SHEET_CUSTOMER);
    var r = findRow(sheet, last4);

    if (action === 'add') {
      if (!r) {
        r = sheet.getLastRow() + 1;
        sheet.getRange(r, 1).setNumberFormat('@').setValue(last4);
        sheet.getRange(r, 2, 1, 4).setValues([['', 0, 0, '']]);
      }
      var stamps = Number(sheet.getRange(r, 3).getValue()) || 0;
      if (stamps >= CFG.GOAL) return full(sheet, r, last4);
      var lastAt = sheet.getRange(r, 5).getValue();
      if (!req.force && lastAt instanceof Date && Date.now() - lastAt.getTime() < 60000) {
        return { ok: false, code: 'RECENT', info: info(sheet, r, last4) };
      }
      stamps++;
      sheet.getRange(r, 3).setValue(stamps);
      sheet.getRange(r, 5).setValue(new Date());
      writeLog(last4, '적립', stamps);
      todayCount(1);
      return done(sheet, r, last4);
    }

    if (!r) return { ok: false, code: 'NOT_FOUND' };
    var cur = Number(sheet.getRange(r, 3).getValue()) || 0;

    if (action === 'undo') {
      if (cur <= 0) return { ok: false, code: 'ZERO', info: info(sheet, r, last4) };
      cur--;
      sheet.getRange(r, 3).setValue(cur);
      sheet.getRange(r, 5).setValue('');         // 방금 적립 기록을 지워서 바로 다시 적립할 수 있게
      writeLog(last4, '취소', cur);
      todayCount(-1);
      return done(sheet, r, last4);
    }

    if (action === 'redeem') {
      if (cur < CFG.GOAL) return { ok: false, code: 'NOT_FULL', info: info(sheet, r, last4) };
      sheet.getRange(r, 3).setValue(0);
      sheet.getRange(r, 4).setValue((Number(sheet.getRange(r, 4).getValue()) || 0) + 1);
      writeLog(last4, '사용 완료', 0);
      return done(sheet, r, last4);
    }
    return { ok: false, code: 'BAD_INPUT' };
  } finally {
    lock.releaseLock();
  }
}

// ------------------------------------------------------------
// 도우미들
// ------------------------------------------------------------
function getSheet(name) {
  var s = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(name);
  if (!s) throw new Error('먼저 처음설정()을 실행해 주세요.');
  return s;
}

function pad4(v) { return ('0000' + String(v)).slice(-4); }

function findRow(sheet, last4) {
  var n = sheet.getLastRow();
  if (n < 2) return 0;
  var vals = sheet.getRange(2, 1, n - 1, 1).getValues();
  for (var i = 0; i < vals.length; i++) {
    if (pad4(vals[i][0]) === last4) return i + 2;
  }
  return 0;
}

function info(sheet, row, last4) {
  var v = sheet.getRange(row, 1, 1, 4).getValues()[0];
  return { last4: last4, nick: String(v[1] || ''), stamps: Number(v[2]) || 0, rewards: Number(v[3]) || 0,
           goal: CFG.GOAL, reward: CFG.REWARD, today: todayCount(0) };
}

function done(sheet, row, last4) { var i = info(sheet, row, last4); i.ok = true; return i; }
function full(sheet, row, last4) { return { ok: false, code: 'FULL', info: info(sheet, row, last4) }; }

function writeLog(last4, what, count) {
  var log = getSheet(SHEET_LOG);
  var r = log.getLastRow() + 1;
  log.getRange(r, 2).setNumberFormat('@');
  log.getRange(r, 1, 1, 4).setValues([[new Date(), last4, what, count]]);
}

// 오늘 적립 횟수 (날짜가 바뀌면 0부터 다시)
function todayCount(delta) {
  var p = PropertiesService.getScriptProperties();
  var d = Utilities.formatDate(new Date(), CFG.TZ, 'yyyy-MM-dd');
  if (p.getProperty('TODAY_DATE') !== d) { p.setProperty('TODAY_DATE', d); p.setProperty('TODAY_COUNT', '0'); }
  var n = Number(p.getProperty('TODAY_COUNT')) || 0;
  if (delta) { n = Math.max(0, n + delta); p.setProperty('TODAY_COUNT', String(n)); }
  return n;
}

// 비밀번호 확인 + 5번 틀리면 10분 잠금
function checkPin(pin) {
  var p = PropertiesService.getScriptProperties();
  var stored = p.getProperty('PIN');
  if (!stored) return { ok: false, code: 'NO_SETUP' };
  var until = Number(p.getProperty('LOCK_UNTIL')) || 0;
  if (Date.now() < until) return { ok: false, code: 'LOCKED', minutes: Math.ceil((until - Date.now()) / 60000) };
  if (String(pin) !== stored) {
    var f = (Number(p.getProperty('FAILS')) || 0) + 1;
    if (f >= 5) { p.setProperty('LOCK_UNTIL', String(Date.now() + 10 * 60000)); f = 0; }
    p.setProperty('FAILS', String(f));
    return { ok: false, code: 'BAD_PIN' };
  }
  p.setProperty('FAILS', '0');
  return null;
}
